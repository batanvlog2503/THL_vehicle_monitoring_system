// Dashboard.jsx
import { useState, useEffect, useRef } from "react"

const Dashboard = () => {
  const canvasRef = useRef(null)
  const wsRef = useRef(null)
  const [detections, setDetections] = useState([])
  const [status, setStatus] = useState("idle")

  const startStream = () => {
    setDetections([])
    setStatus("streaming")

    const ws = new WebSocket("ws://localhost:8765")
    wsRef.current = ws

    ws.onmessage = (event) => {
      const data = JSON.parse(event.data)

      if (data.status === "done") {
        setStatus("done")
        return
      }

      // Vẽ frame lên canvas (bbox đã được YOLO vẽ sẵn trong Python)
      const canvas = canvasRef.current
      const ctx = canvas.getContext("2d")
      const img = new Image()
      img.src = `data:image/jpeg;base64,${data.frame}`
      img.onload = () => {
        canvas.width = img.width
        canvas.height = img.height
        ctx.drawImage(img, 0, 0)
      }

      // Cập nhật bảng — giữ lại tất cả ID đã thấy
      setDetections((prev) => {
        const updated = [...prev]
        data.detections.forEach((d) => {
          const idx = updated.findIndex((x) => x.id === d.id)
          if (idx === -1)
            updated.push(d) // ID mới → thêm
          else updated[idx] = d // đã có → cập nhật
        })
        return updated
      })
    }

    ws.onclose = () => setStatus("idle")
  }

  const stopStream = () => {
    wsRef.current?.close()
    setStatus("idle")
  }

  return (
    <div className="container-fluid dashboard">
      <div className="row">
        {/* Canvas video */}
        <div className="col-12 col-md-8">
          <canvas
            ref={canvasRef}
            style={{ width: "100%", background: "#000" }}
          />
          <div className="mt-2 d-flex gap-2">
            <button
              className="btn btn-success"
              onClick={startStream}
              disabled={status === "streaming"}
            >
              ▶ Start
            </button>
            <button
              className="btn btn-danger"
              onClick={stopStream}
              disabled={status !== "streaming"}
            >
              ⏹ Stop
            </button>
            <span
              className={`badge my-auto ${
                status === "streaming"
                  ? "bg-success"
                  : status === "done"
                    ? "bg-secondary"
                    : "bg-warning"
              }`}
            >
              {status === "streaming"
                ? "🔴 LIVE"
                : status === "done"
                  ? "✅ Done"
                  : "⏸ Idle"}
            </span>
          </div>
        </div>

        {/* Bảng detection */}
        <div className="col-12 col-md-4">
          <h5>Live Detection — {detections.length} objects</h5>
          <div style={{ overflowY: "auto", maxHeight: "80vh" }}>
            <table className="table table-sm table-hover">
              <thead className="table-dark sticky-top">
                <tr>
                  <th>ID</th>
                  <th>Class</th>
                  <th>Confidence</th>
                </tr>
              </thead>
              <tbody>
                {detections.length === 0 ? (
                  <tr>
                    <td
                      colSpan={3}
                      className="text-center text-muted py-3"
                    >
                      {status === "streaming"
                        ? "🔍 Đang phân tích..."
                        : "Chưa có dữ liệu"}
                    </td>
                  </tr>
                ) : (
                  detections.map((d) => (
                    <tr key={d.id}>
                      <td>
                        <span className="badge bg-primary">{d.id}</span>
                      </td>
                      <td>{d.label}</td>
                      <td>
                        <div
                          className="progress"
                          style={{ height: "6px" }}
                        >
                          <div
                            className={`progress-bar ${d.conf > 0.7 ? "bg-success" : "bg-warning"}`}
                            style={{ width: `${d.conf * 100}%` }}
                          />
                        </div>
                        <small>{(d.conf * 100).toFixed(0)}%</small>
                      </td>
                    </tr>
                  ))
                )}
              </tbody>
            </table>
          </div>
        </div>
      </div>
    </div>
  )
}

export default Dashboard
