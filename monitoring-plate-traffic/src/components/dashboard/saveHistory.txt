🚀 server.js
const express = require("express")
const mongoose = require("mongoose")
const cors = require("cors")

const app = express()

app.use(cors())
app.use(express.json({ limit: "50mb" }))

// ===== CONNECT MONGO =====
mongoose.connect("mongodb://127.0.0.1:27017/yolo", {
  useNewUrlParser: true,
  useUnifiedTopology: true,
})

console.log("MongoDB connected")

// ===== SCHEMA =====
const DetectionSchema = new mongoose.Schema({
  frame: Number,
  id: Number,
  label: String,
  conf: Number,
  bbox: [Number],
})

const HistorySchema = new mongoose.Schema({
  videoName: String,
  createdAt: { type: Date, default: Date.now },
  detections: [DetectionSchema],
})

const History = mongoose.model("History", HistorySchema)

// ===== API SAVE =====
app.post("/api/save-history", async (req, res) => {
  try {
    const { detections, videoName } = req.body

    const history = new History({
      videoName,
      detections,
    })

    await history.save()

    console.log("Saved history:", history._id)

    res.json({ success: true })
  } catch (err) {
    console.error(err)
    res.status(500).json({ error: "Save failed" })
  }
})

// ===== API GET =====
app.get("/api/history", async (req, res) => {
  const data = await History.find().sort({ createdAt: -1 })
  res.json(data)
})

app.listen(3000, () => {
  console.log("NodeJS running at http://localhost:3000")
})
