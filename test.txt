import express from 'express';
import { engine } from 'express-handlebars';

const app = express();

app.engine('handlebars', engine());
app.set('view engine', 'handlebars');
app.set('views', './views');

app.get('/', (req, res) => {
    res.render('home');
});

app.listen(3000);

-- middleware
🧠 Middleware là gì?

Middleware là một hàm đứng ở giữa request và response

Luồng xử lý:

Client → Middleware → Middleware → Route → Response


Middleware có thể:

Xử lý request

Thay đổi request / response

Dừng request

Chuyển sang middleware tiếp theo

✋ Middleware làm được những gì?

Middleware có thể:

Kiểm tra đăng nhập

Validate dữ liệu

Parse body

Ghi log

Bắt lỗi

Phân quyền
