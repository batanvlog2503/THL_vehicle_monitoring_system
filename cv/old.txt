# stream.py
import cv2, json, base64, asyncio, websockets
from ultralytics import YOLO

model = YOLO("yolov8n.pt")

async def stream(websocket):
    print("Client kết nối!")
    cap = cv2.VideoCapture("sample.mp4")

    while True:
        ret, frame = cap.read()
        if not ret:
            await websocket.send(json.dumps({'status': 'done'})) # hết video trả done
            break

        # ── Phần YOLO của bạn, giữ nguyên ──
        results = model.track(frame, persist=True)
        r = results[0]
        frame_ = r.plot()  # frame đã vẽ bbox sẵn

        boxes = r.boxes # id, cls, conf, bbox
        detections = []

        if boxes.id is not None:
            ids     = boxes.id.cpu().numpy().astype(int)
            classes = boxes.cls.cpu().numpy().astype(int)
            confs   = boxes.conf.cpu().numpy()
            bboxes  = boxes.xyxy.cpu().numpy()

            for obj_id, cls, conf, bbox in zip(ids, classes, confs, bboxes):
                label = model.names[cls]
                print(f"ID:{obj_id} Class:{label} Conf:{conf:.2f}")

                detections.append({
                    'id':    int(obj_id),
                    'label': label,
                    'conf':  round(float(conf), 2),
                    'bbox':  bbox.tolist()
                }) # đống gói json gửi vào trong detections

        # ─Encode frame đã vẽ bbox → gửi sang React ──
        _, buffer = cv2.imencode('.jpg', frame_, [cv2.IMWRITE_JPEG_QUALITY, 60])
        frame_b64 = base64.b64encode(buffer).decode('utf-8')

        try:
            await websocket.send(json.dumps({
                'status':     'frame',
                'frame':      frame_b64,
                'detections': detections
            }))
        except websockets.exceptions.ConnectionClosed:
            break

        await asyncio.sleep(1/25) # 25fps

    cap.release()

async def main():
    async with websockets.serve(stream, 'localhost', 8765):
        print("WebSocket chạy tại ws://localhost:8765")
        await asyncio.Future()

asyncio.run(main())
